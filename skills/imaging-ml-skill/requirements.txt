# requirements.txt for imaging-ml-skill — pip fallback (base / demo stack).

# Supported path is uv + uv.lock (see references/environment_setup.md). Use this only if uv
# is unavailable. Python 3.11. pyradiomics 3.0.1 needs NumPy 1.x, an older setuptools, and
# versioneer at build time, so install in this order in a clean venv:
#
#   python3.11 -m venv .venv && source .venv/bin/activate
#   pip install --upgrade pip
#   pip install "setuptools<65" wheel versioneer "numpy==1.26.4"
#   pip install --no-build-isolation -r requirements.txt
#
# For the real-data (IDC) path, then:
#   pip install --no-build-isolation -r requirements-real.txt
#
# Why the pins: numpy<2 (pyradiomics 3.0.1's C extension is NumPy-1.x ABI and will not import
# under NumPy 2.x) and pandas 2.2.x (so idc-index, which needs pandas<=2.2.4, can coexist).
# See references/environment_setup.md for the reproduced failures.
